template <class T> int pick(T value) {
  if constexpr (sizeof(T) == 1) {
    return 1;
#ifndef NO_WIDE
  } else if constexpr (sizeof(T) == 2) {
    return 2;
#else
  } else {
    return 0;
#endif
  }
  return value;
}
